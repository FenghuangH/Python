# TODO: implement normalize(v)
def normalize(v: np.ndarray) -> np.ndarray:
    v = np.asarray(v, dtype=float)
    low, high = v.min(), v.max()
    v = (v - low) / (high - low) if high > low else np.zeros_like(v)
    return v
